### 8.1 Сборщики и препроцессоры

**Настройка Vite для компонентного подхода:**

```javascript
// vite.config.js
import { defineConfig } from 'vite';
import { resolve } from 'path';

export default defineConfig({
	root: 'src',
	build: {
		outDir: '../dist',
		rollupOptions: {
			input: {
				main: resolve(__dirname, 'src/index.html'),
				about: resolve(__dirname, 'src/about.html'),
				contact: resolve(__dirname, 'src/contact.html'),
			},
		},
	},
	css: {
		preprocessorOptions: {
			scss: {
				additionalData: `
					@import "styles/abstracts/variables";
					@import "styles/abstracts/mixins";
					@import "styles/abstracts/functions";
				`,
			},
		},
	},
	resolve: {
		alias: {
			'@': resolve(__dirname, 'src'),
			'@components': resolve(__dirname, 'src/components'),
			'@styles': resolve(__dirname, 'src/styles'),
		},
	},
});
```

### 8.2 Система сборки компонентов

**ComponentBuilder.js - автоматическая сборка компонентов:**

```javascript
// scripts/build/ComponentBuilder.js
import fs from 'fs/promises';
import path from 'path';
import { glob } from 'glob';

class ComponentBuilder {
	constructor(config = {}) {
		this.config = {
			componentsDir: 'src/components',
			outputDir: 'dist',
			templatesDir: 'src/templates',
			...config,
		};

		this.components = new Map();
		this.templates = new Map();
	}

	async build() {
		console.log('🔨 Начинаем сборку компонентов...');

		// Загружаем все компоненты
		await this.loadComponents();

		// Загружаем шаблоны
		await this.loadTemplates();

		// Собираем страницы
		await this.buildPages();

		console.log('✅ Сборка завершена!');
	}

	async loadComponents() {
		const componentPaths = await glob(`${this.config.componentsDir}/**/*.html`);

		for (const componentPath of componentPaths) {
			const component = await this.loadComponent(componentPath);
			if (component) {
				this.components.set(component.name, component);
			}
		}

		console.log(`📦 Загружено ${this.components.size} компонентов`);
	}

	async loadComponent(htmlPath) {
		try {
			const componentDir = path.dirname(htmlPath);
			const componentName = path.basename(componentDir);

			// Загружаем HTML
			const html = await fs.readFile(htmlPath, 'utf-8');

			// Ищем SCSS файл
			const scssPath = path.join(componentDir, `${componentName}.scss`);
			let scss = '';
			try {
				scss = await fs.readFile(scssPath, 'utf-8');
			} catch (e) {
				// SCSS необязателен
			}

			// Ищем JS файл
			const jsPath = path.join(componentDir, `${componentName}.js`);
			let js = '';
			try {
				js = await fs.readFile(jsPath, 'utf-8');
			} catch (e) {
				// JS необязателен
			}

			return {
				name: componentName,
				html,
				scss,
				js,
				path: componentDir,
			};
		} catch (error) {
			console.error(`❌ Ошибка загрузки компонента ${htmlPath}:`, error);
			return null;
		}
	}

	async buildPages() {
		const pagePaths = await glob('src/pages/**/*.html');

		for (const pagePath of pagePaths) {
			await this.buildPage(pagePath);
		}
	}

	async buildPage(pagePath) {
		try {
			let html = await fs.readFile(pagePath, 'utf-8');

			// Обрабатываем включения компонентов
			html = this.processIncludes(html);

			// Обрабатываем шаблоны
			html = this.processTemplates(html);

			// Сохраняем результат
			const outputPath = path.join(
				this.config.outputDir,
				path.relative('src/pages', pagePath)
			);

			await fs.mkdir(path.dirname(outputPath), { recursive: true });
			await fs.writeFile(outputPath, html);

			console.log(`📄 Собрана страница: ${path.basename(pagePath)}`);
		} catch (error) {
			console.error(`❌ Ошибка сборки страницы ${pagePath}:`, error);
		}
	}

	processIncludes(html) {
		// Обрабатываем теги вида <include src="component-name" />
		return html.replace(
			/<include\s+src="([^"]+)"([^>]*?)\/>/g,
			(match, componentName, attributes) => {
				const component = this.components.get(componentName);
				if (!component) {
					console.warn(`⚠️ Компонент "${componentName}" не найден`);
					return match;
				}

				// Парсим атрибуты
				const props = this.parseAttributes(attributes);

				// Подставляем данные в компонент
				return this.renderComponent(component, props);
			}
		);
	}

	parseAttributes(attributesString) {
		const props = {};
		const regex = /(\w+)="([^"]*)"/g;
		let match;

		while ((match = regex.exec(attributesString)) !== null) {
			props[match[1]] = match[2];
		}

		return props;
	}

	renderComponent(component, props = {}) {
		let html = component.html;

		// Заменяем переменные вида {{propName}}
		Object.entries(props).forEach(([key, value]) => {
			const regex = new RegExp(`{{\\s*${key}\\s*}}`, 'g');
			html = html.replace(regex, value);
		});

		// Очищаем незаменённые переменные
		html = html.replace(/{{[^}]+}}/g, '');

		return html;
	}
}

export default ComponentBuilder;
```

### 8.3 Автоматизация с Gulp

**gulpfile.js для сложных проектов:**

```javascript
// gulpfile.js
import gulp from 'gulp';
import sass from 'gulp-sass';
import * as dartSass from 'sass';
import postcss from 'gulp-postcss';
import autoprefixer from 'autoprefixer';
import cssnano from 'cssnano';
import uglify from 'gulp-uglify';
import imagemin from 'gulp-imagemin';
import svgSprite from 'gulp-svg-sprite';
import browserSync from 'browser-sync';
import ComponentBuilder from './scripts/build/ComponentBuilder.js';

const sassCompiler = sass(dartSass);
const bs = browserSync.create();

// Пути
const paths = {
	styles: {
		src: 'src/styles/**/*.scss',
		dest: 'dist/css',
	},
	scripts: {
		src: 'src/scripts/**/*.js',
		dest: 'dist/js',
	},
	images: {
		src: 'src/images/**/*',
		dest: 'dist/images',
	},
	components: {
		src: 'src/components/**/*',
		dest: 'dist',
	},
};

// Компиляция SCSS
export function styles() {
	return gulp
		.src(paths.styles.src)
		.pipe(sassCompiler().on('error', sassCompiler.logError))
		.pipe(postcss([autoprefixer(), cssnano()]))
		.pipe(gulp.dest(paths.styles.dest))
		.pipe(bs.stream());
}

// Сборка компонентов
export async function components() {
	const builder = new ComponentBuilder();
	await builder.build();
}

// Обработка изображений
export function images() {
	return gulp
		.src(paths.images.src)
		.pipe(
			imagemin([
				imagemin.gifsicle({ interlaced: true }),
				imagemin.mozjpeg({ quality: 80, progressive: true }),
				imagemin.optipng({ optimizationLevel: 5 }),
				imagemin.svgo({
					plugins: [{ removeViewBox: false }, { cleanupIDs: false }],
				}),
			])
		)
		.pipe(gulp.dest(paths.images.dest));
}

// Создание SVG спрайта
export function svgSprites() {
	return gulp
		.src('src/icons/**/*.svg')
		.pipe(
			svgSprite({
				mode: {
					symbol: {
						sprite: '../sprite.svg',
						example: false,
					},
				},
			})
		)
		.pipe(gulp.dest('dist/images'));
}

// Обработка скриптов
export function scripts() {
	return gulp
		.src(paths.scripts.src)
		.pipe(uglify())
		.pipe(gulp.dest(paths.scripts.dest))
		.pipe(bs.stream());
}

// Сервер разработки
export function serve(done) {
	bs.init({
		server: {
			baseDir: 'dist',
		},
		port: 3000,
	});
	done();
}

// Отслеживание изменений
export function watch() {
	gulp.watch(paths.styles.src, styles);
	gulp.watch(paths.scripts.src, scripts);
	gulp.watch('src/components/**/*', components);
	gulp.watch('src/pages/**/*.html', components);
	gulp.watch('dist/**/*.html').on('change', bs.reload);
}

// Основные команды
export const build = gulp.series(
	gulp.parallel(styles, scripts, images, svgSprites),
	components
);

export const dev = gulp.series(build, serve, watch);

export default dev;
```
