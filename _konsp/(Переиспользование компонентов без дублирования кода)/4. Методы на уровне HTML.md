## 4. Методы на уровне HTML

### 4.1 HTML Templates с тегом `<template>`

**HTML5 template** элемент позволяет создавать переиспользуемые фрагменты HTML:

```html
<!-- Шаблон карточки товара -->
<template id="product-card-template">
	<article class="card card--product">
		<div class="card__image-wrapper">
			<img class="card__image" src="" alt="" loading="lazy" />
			<span class="card__badge"></span>
		</div>
		<div class="card__content">
			<h3 class="card__title"></h3>
			<p class="card__description"></p>
			<div class="card__price">
				<span class="card__price-current"></span>
				<span class="card__price-old"></span>
			</div>
			<div class="card__actions">
				<button class="button button--primary card__buy-btn">Купить</button>
				<button class="button button--outline card__favorite-btn">
					<svg class="button__icon">
						<!-- Иконка сердечка -->
					</svg>
				</button>
			</div>
		</div>
	</article>
</template>

<!-- JavaScript для использования шаблона -->
<script>
	function createProductCard(productData) {
		// Клонируем шаблон
		const template = document.getElementById('product-card-template');
		const clone = template.content.cloneNode(true);

		// Заполняем данными
		clone.querySelector('.card__image').src = productData.image;
		clone.querySelector('.card__image').alt = productData.title;
		clone.querySelector('.card__title').textContent = productData.title;
		clone.querySelector('.card__description').textContent =
			productData.description;
		clone.querySelector('.card__price-current').textContent = productData.price;

		// Если есть старая цена
		if (productData.oldPrice) {
			clone.querySelector('.card__price-old').textContent =
				productData.oldPrice;
		}

		// Если есть скидка
		if (productData.badge) {
			clone.querySelector('.card__badge').textContent = productData.badge;
		}

		return clone;
	}

	// Использование
	const products = [
		{
			title: 'iPhone 15 Pro',
			description: 'Новейший смартфон от Apple',
			price: '120 000 ₽',
			oldPrice: '130 000 ₽',
			badge: '-8%',
			image: '/images/iphone15.jpg',
		},
		// ... другие товары
	];

	const container = document.querySelector('.products-grid');
	products.forEach((product) => {
		const card = createProductCard(product);
		container.appendChild(card);
	});
</script>
```

### 4.2 Web Components (Custom Elements)

**Web Components** - стандартная технология для создания переиспользуемых компонентов:

```html
<!-- Определение пользовательского элемента -->
<script>
	class ProductCard extends HTMLElement {
		constructor() {
			super();

			// Создаем Shadow DOM для инкапсуляции стилей
			this.attachShadow({ mode: 'open' });

			// Шаблон компонента
			this.shadowRoot.innerHTML = `
			<style>
				:host {
					display: block;
					margin-bottom: 24px;
				}
				
				.card {
					background: white;
					border-radius: 12px;
					box-shadow: 0 2px 16px rgba(0, 0, 0, 0.1);
					overflow: hidden;
					transition: transform 0.2s ease;
				}
				
				.card:hover {
					transform: translateY(-4px);
				}
				
				.card__image {
					width: 100%;
					height: 200px;
					object-fit: cover;
				}
				
				.card__content {
					padding: 20px;
				}
				
				.card__title {
					font-size: 1.25rem;
					font-weight: 600;
					margin: 0 0 8px 0;
					color: #333;
				}
				
				.card__description {
					color: #666;
					margin: 0 0 16px 0;
					line-height: 1.5;
				}
				
				.card__price {
					display: flex;
					align-items: center;
					gap: 8px;
					margin-bottom: 16px;
				}
				
				.card__price-current {
					font-size: 1.5rem;
					font-weight: 700;
					color: #007bff;
				}
				
				.card__price-old {
					text-decoration: line-through;
					color: #999;
				}
				
				.card__actions {
					display: flex;
					gap: 12px;
				}
				
				.button {
					padding: 12px 24px;
					border: none;
					border-radius: 8px;
					font-weight: 500;
					cursor: pointer;
					transition: all 0.2s ease;
				}
				
				.button--primary {
					background: #007bff;
					color: white;
					flex: 1;
				}
				
				.button--primary:hover {
					background: #0056b3;
				}
				
				.button--outline {
					background: transparent;
					border: 2px solid #007bff;
					color: #007bff;
					width: 48px;
					height: 48px;
				}
				
				.badge {
					position: absolute;
					top: 12px;
					right: 12px;
					background: #dc3545;
					color: white;
					padding: 4px 8px;
					border-radius: 4px;
					font-size: 0.875rem;
					font-weight: 600;
				}
				
				.image-wrapper {
					position: relative;
				}
			</style>
			
			<article class="card">
				<div class="image-wrapper">
					<img class="card__image" loading="lazy">
					<span class="badge" style="display: none;"></span>
				</div>
				<div class="card__content">
					<h3 class="card__title"></h3>
					<p class="card__description"></p>
					<div class="card__price">
						<span class="card__price-current"></span>
						<span class="card__price-old" style="display: none;"></span>
					</div>
					<div class="card__actions">
						<button class="button button--primary">
							Купить
						</button>
						<button class="button button--outline">
							♥
						</button>
					</div>
				</div>
			</article>
		`;

			this.render();
		}

		// Атрибуты, за изменениями которых нужно следить
		static get observedAttributes() {
			return ['title', 'description', 'price', 'old-price', 'image', 'badge'];
		}

		// Вызывается при изменении атрибутов
		attributeChangedCallback(name, oldValue, newValue) {
			this.render();
		}

		render() {
			const title = this.getAttribute('title') || '';
			const description = this.getAttribute('description') || '';
			const price = this.getAttribute('price') || '';
			const oldPrice = this.getAttribute('old-price');
			const image = this.getAttribute('image') || '';
			const badge = this.getAttribute('badge');

			// Обновляем содержимое
			this.shadowRoot.querySelector('.card__title').textContent = title;
			this.shadowRoot.querySelector('.card__description').textContent =
				description;
			this.shadowRoot.querySelector('.card__price-current').textContent = price;
			this.shadowRoot.querySelector('.card__image').src = image;
			this.shadowRoot.querySelector('.card__image').alt = title;

			// Старая цена (если есть)
			const oldPriceEl = this.shadowRoot.querySelector('.card__price-old');
			if (oldPrice) {
				oldPriceEl.textContent = oldPrice;
				oldPriceEl.style.display = 'inline';
			} else {
				oldPriceEl.style.display = 'none';
			}

			// Бейдж скидки (если есть)
			const badgeEl = this.shadowRoot.querySelector('.badge');
			if (badge) {
				badgeEl.textContent = badge;
				badgeEl.style.display = 'block';
			} else {
				badgeEl.style.display = 'none';
			}
		}
	}

	// Регистрируем пользовательский элемент
	customElements.define('product-card', ProductCard);
</script>

<!-- Использование Web Component -->
<div class="products-grid">
	<product-card
		title="iPhone 15 Pro"
		description="Новейший смартфон от Apple"
		price="120 000 ₽"
		old-price="130 000 ₽"
		badge="-8%"
		image="/images/iphone15.jpg"
	>
	</product-card>

	<product-card
		title="Samsung Galaxy S24"
		description="Флагман от Samsung"
		price="90 000 ₽"
		image="/images/galaxy-s24.jpg"
	>
	</product-card>
</div>
```

### 4.4 Статические генераторы сайтов

**Пример с использованием простой системы сборки:**

```javascript
// build.js - простой сборщик
const fs = require('fs');
const path = require('path');

// Функция для загрузки компонента
function loadComponent(name) {
	const componentPath = path.join(__dirname, 'components', `${name}.html`);
	return fs.readFileSync(componentPath, 'utf8');
}

// Функция для обработки includes
function processIncludes(content) {
	return content.replace(/{{include\s+(.+?)}}/g, (match, componentName) => {
		return loadComponent(componentName.trim());
	});
}

// Функция для замены переменных
function processVariables(content, variables = {}) {
	return content.replace(/{{(.+?)}}/g, (match, varName) => {
		const trimmedName = varName.trim();
		return variables[trimmedName] || match;
	});
}

// Сборка страницы
function buildPage(templatePath, outputPath, variables = {}) {
	let content = fs.readFileSync(templatePath, 'utf8');

	// Обрабатываем includes
	content = processIncludes(content);

	// Обрабатываем переменные
	content = processVariables(content, variables);

	// Сохраняем результат
	fs.writeFileSync(outputPath, content);
	console.log(`Собрана страница: ${outputPath}`);
}

// Сборка всех страниц
const pages = [
	{
		template: 'src/index.html',
		output: 'dist/index.html',
		variables: {
			title: 'Главная страница',
			description: 'Добро пожаловать на наш сайт',
		},
	},
	{
		template: 'src/about.html',
		output: 'dist/about.html',
		variables: {
			title: 'О нас',
			description: 'Информация о нашей компании',
		},
	},
];

pages.forEach((page) => {
	buildPage(page.template, page.output, page.variables);
});
```

```html
<!-- components/header.html -->

	</head>

		<main>
			<h1>{{title}}</h1>
			<p>{{description}}</p>
		</main>

		{{include footer}}
	</body>
</html>
```
