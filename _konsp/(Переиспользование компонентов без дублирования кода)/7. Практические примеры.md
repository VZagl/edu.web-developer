### 7.1 Адаптивные кнопки

**Полнофункциональная система кнопок:**

```html
<!-- HTML разметка -->
<div class="buttons-demo">
	<!-- Базовые варианты -->
	<button class="btn btn--primary">Основная кнопка</button>
	<button class="btn btn--secondary">Вторичная кнопка</button>
	<button class="btn btn--outline">Прозрачная кнопка</button>

	<!-- Размеры -->
	<button class="btn btn--primary btn--small">Маленькая</button>
	<button class="btn btn--primary btn--medium">Средняя</button>
	<button class="btn btn--primary btn--large">Большая</button>

	<!-- С иконками -->
	<button class="btn btn--primary">
		<svg class="btn__icon btn__icon--left" width="16" height="16">
			<use href="#icon-download"></use>
		</svg>
		<span class="btn__text">Скачать</span>
	</button>

	<!-- Адаптивная кнопка -->
	<button class="btn btn--responsive btn--primary">
		<svg class="btn__icon btn__icon--mobile" width="16" height="16">
			<use href="#icon-menu"></use>
		</svg>
		<span class="btn__text btn__text--desktop">Открыть меню</span>
	</button>

	<!-- Состояния -->
	<button class="btn btn--primary" disabled>Отключена</button>
	<button class="btn btn--primary btn--loading">
		<span class="btn__text">Загрузка...</span>
	</button>
</div>
```
