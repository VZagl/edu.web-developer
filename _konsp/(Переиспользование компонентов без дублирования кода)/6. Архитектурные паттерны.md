```javascript
// components/atoms/Button/Button.js
import './Button.scss';

class Button {
	constructor(element, options = {}) {
		this.element = element;
		this.options = {
			variant: 'primary',
			size: 'medium',
			disabled: false,
			loading: false,
			...options,
		};

		this.init();
	}

	init() {
		this.updateClasses();
		this.bindEvents();
	}

	updateClasses() {
		const { variant, size, disabled, loading } = this.options;

		// Базовый класс
		this.element.classList.add('btn');

		// Модификаторы
		if (variant) this.element.classList.add(`btn--${variant}`);
		if (size) this.element.classList.add(`btn--${size}`);
		if (disabled) this.element.classList.add('btn--disabled');
		if (loading) this.element.classList.add('btn--loading');
	}

	bindEvents() {
		this.element.addEventListener('click', (e) => {
			if (this.options.disabled || this.options.loading) {
				e.preventDefault();
				return;
			}

			// Эмитируем кастомное событие
			this.element.dispatchEvent(
				new CustomEvent('btn:click', {
					detail: { button: this },
				})
			);
		});
	}

	setLoading(loading) {
		this.options.loading = loading;
		this.element.classList.toggle('btn--loading', loading);
	}

	setDisabled(disabled) {
		this.options.disabled = disabled;
		this.element.classList.toggle('btn--disabled', disabled);
		this.element.disabled = disabled;
	}

	// Статический метод для автоинициализации
	static autoInit() {
		const buttons = document.querySelectorAll('[data-component="button"]');
		return Array.from(buttons).map((el) => {
			const options = JSON.parse(el.dataset.options || '{}');
			return new Button(el, options);
		});
	}
}

export default Button;
```

// components/atoms/Input/Input.js
class Input {
	constructor(element, options = {}) {
		this.element = element;
		this.options = {
			type: 'text',
			placeholder: '',
			required: false,
			validation: null,
			...options,
		};

		this.init();
	}

	init() {
		this.setupValidation();
		this.bindEvents();
	}

	setupValidation() {
		if (this.options.validation) {
			this.element.addEventListener('blur', () => this.validate());
		}
	}

	validate() {
		const value = this.getValue();
		const isValid = this.options.validation(value);

		this.element.classList.toggle('input--error', !isValid);

		// Эмитируем событие валидации
		this.element.dispatchEvent(
			new CustomEvent('input:validate', {
				detail: { value, isValid },
			})
		);

		return isValid;
	}

	getValue() {
		return this.element.value;
	}

	setValue(value) {
		this.element.value = value;
	}

	bindEvents() {
		this.element.addEventListener('input', (e) => {
			this.element.dispatchEvent(
				new CustomEvent('input:change', {
					detail: { value: e.target.value },
				})
			);
		});
	}
}

// components/molecules/FormField/FormField.js
class FormField {
	constructor(element, options = {}) {
		this.element = element;
		this.options = options;

		// Находим дочерние элементы
		this.label = element.querySelector('.label');
		this.input = element.querySelector('.input');
		this.errorElement = element.querySelector('.form-field__error');

		// Инициализируем Input компонент
		if (this.input) {
			this.inputComponent = new Input(this.input, {
				validation: this.options.validation,
			});
		}

		this.init();
	}

	init() {
		this.bindEvents();
	}

	bindEvents() {
		if (this.input) {
			// Слушаем события от Input компонента
			this.input.addEventListener('input:validate', (e) => {
				this.showError(!e.detail.isValid);
			});
		}
	}

	showError(hasError, message = '') {
		this.element.classList.toggle('form-field--error', hasError);

		if (this.errorElement) {
			this.errorElement.textContent = hasError ? message : '';
			this.errorElement.style.display = hasError ? 'block' : 'none';
		}
	}

	getValue() {
		return this.inputComponent ? this.inputComponent.getValue() : '';
	}

	setValue(value) {
		if (this.inputComponent) {
			this.inputComponent.setValue(value);
		}
	}
}

// components/organisms/ContactForm/ContactForm.js
class ContactForm {
	constructor(element, options = {}) {
		this.element = element;
		this.options = options;

		// Инициализируем все FormField компоненты
		this.fields = {};
		this.initFields();

		this.init();
	}

	initFields() {
		const fieldElements = this.element.querySelectorAll('.form-field');

		fieldElements.forEach((fieldEl) => {
			const fieldName = fieldEl.dataset.field;
			if (fieldName) {
				this.fields[fieldName] = new FormField(fieldEl, {
					validation: this.getValidationForField(fieldName),
				});
			}
		});
	}

	getValidationForField(fieldName) {
		const validations = {
			email: (value) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value),
			phone: (value) => /^\+?[\d\s\-\(\)]{10,}$/.test(value),
			name: (value) => value.length >= 2,
		};

		return validations[fieldName] || (() => true);
	}

	init() {
		this.bindEvents();
	}

	bindEvents() {
		this.element.addEventListener('submit', (e) => {
			e.preventDefault();
			this.handleSubmit();
		});
	}

	async handleSubmit() {
		// Валидируем все поля
		const isValid = this.validateAll();

		if (!isValid) {
			this.showFormError('Пожалуйста, исправьте ошибки в форме');
			return;
		}

		// Получаем данные формы
		const formData = this.getFormData();

		try {
			// Отправляем данные
			const response = await fetch('/api/contact', {
				method: 'POST',
				headers: { 'Content-Type': 'application/json' },
				body: JSON.stringify(formData),
			});

			if (response.ok) {
				this.showSuccess('Сообщение отправлено успешно!');
				this.resetForm();
			} else {
				throw new Error('Ошибка отправки');
			}
		} catch (error) {
			this.showFormError('Произошла ошибка при отправке');
		}
	}

	validateAll() {
		let isValid = true;

		Object.values(this.fields).forEach((field) => {
			if (field.inputComponent && !field.inputComponent.validate()) {
				isValid = false;
			}
		});

		return isValid;
	}

	getFormData() {
		const data = {};

		Object.entries(this.fields).forEach(([name, field]) => {
			data[name] = field.getValue();
		});

		return data;
	}

	resetForm() {
		Object.values(this.fields).forEach((field) => {
			field.setValue('');
			field.showError(false);
		});
	}

	showSuccess(message) {
		// Показываем сообщение об успехе
		console.log('Success:', message);
	}

	showFormError(message) {
		// Показываем ошибку формы
		console.log('Error:', message);
	}
}

**Автоинициализация компонентов:**

```javascript
// scripts/app.js - инициализация всех компонентов
import { Button, Input, SearchBox, ContactForm, Header } from '../components';

class App {
	constructor() {
		this.components = [];
		this.init();
	}

	init() {
		// Инициализируем компоненты по data-атрибутам
		this.initComponentsByAttribute();

		// Глобальные слушатели
		this.bindGlobalEvents();
	}

	initComponentsByAttribute() {
		const componentMap = {
			button: Button,
			'search-box': SearchBox,
			'contact-form': ContactForm,
			header: Header,
		};

		Object.entries(componentMap).forEach(([name, ComponentClass]) => {
			const elements = document.querySelectorAll(`[data-component="${name}"]`);

			elements.forEach((element) => {
				const options = this.parseOptions(element);
				const component = new ComponentClass(element, options);
				this.components.push(component);
			});
		});
	}

	parseOptions(element) {
		try {
			return JSON.parse(element.dataset.options || '{}');
		} catch (e) {
			console.warn('Неверный JSON в data-options:', element);
			return {};
		}
	}

	bindGlobalEvents() {
		// Глобальная обработка кликов для аналитики
		document.addEventListener('click', (e) => {
			const component = e.target.closest('[data-component]');
			if (component) {
				this.trackComponentInteraction(component, e);
			}
		});
	}

	trackComponentInteraction(component, event) {
		const componentType = component.dataset.component;
		const action = event.target.dataset.action || 'click';

		// Отправляем аналитику
		if (window.gtag) {
			window.gtag('event', action, {
				component_type: componentType,
				component_id: component.id || 'unnamed',
			});
		}
	}

	// Метод для уничтожения всех компонентов (для SPA)
	destroy() {
		this.components.forEach((component) => {
			if (component.destroy) {
				component.destroy();
			}
		});
		this.components = [];
	}
}

// Инициализируем приложение
document.addEventListener('DOMContentLoaded', () => {
	window.app = new App();
});
```
