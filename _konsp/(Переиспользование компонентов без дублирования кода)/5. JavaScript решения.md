### 5.1 Создание системы компонентов

**Базовый класс компонента:**

```javascript
// ComponentBase.js - базовый класс для всех компонентов
class ComponentBase {
	constructor(element, options = {}) {
		this.element = element;
		this.options = { ...this.defaultOptions, ...options };
		this.state = {};

		this.init();
		this.bindEvents();
	}

	// Настройки по умолчанию (переопределяются в наследниках)
	get defaultOptions() {
		return {};
	}

	// Инициализация компонента
	init() {
		this.element.classList.add(this.constructor.cssClass);
	}

	// Привязка событий
	bindEvents() {
		// Переопределяется в наследниках
	}

	// Рендеринг компонента
	render() {
		// Переопределяется в наследниках
	}

	// Обновление состояния
	setState(newState) {
		this.state = { ...this.state, ...newState };
		this.render();
	}

	// Уничтожение компонента
	destroy() {
		// Удаляем слушатели событий
		this.element.removeEventListener('click', this.handleClick);
		// Очищаем ссылки
		this.element = null;
		this.options = null;
		this.state = null;
	}

	// Статический метод для инициализации компонентов
	static initAll(selector, options = {}) {
		const elements = document.querySelectorAll(selector);
		return Array.from(elements).map((el) => new this(el, options));
	}
}
```

**Пример компонента - Модальное окно:**

```javascript
// Modal.js - компонент модального окна
class Modal extends ComponentBase {
	static cssClass = 'modal';

	get defaultOptions() {
		return {
			closeOnBackdrop: true,
			closeOnEscape: true,
			showCloseButton: true,
			animation: 'fade',
		};
	}

	init() {
		super.init();

		// Создаем структуру модального окна
		this.element.innerHTML = `
			<div class="modal__backdrop"></div>
			<div class="modal__container">
				<div class="modal__header">
					<h2 class="modal__title"></h2>
					${
						this.options.showCloseButton
							? '<button class="modal__close" aria-label="Закрыть">&times;</button>'
							: ''
					}
				</div>
				<div class="modal__body"></div>
				<div class="modal__footer"></div>
			</div>
		`;

		// Кешируем элементы
		this.backdrop = this.element.querySelector('.modal__backdrop');
		this.container = this.element.querySelector('.modal__container');
		this.title = this.element.querySelector('.modal__title');
		this.body = this.element.querySelector('.modal__body');
		this.footer = this.element.querySelector('.modal__footer');
		this.closeBtn = this.element.querySelector('.modal__close');

		// Устанавливаем начальное состояние
		this.setState({ isOpen: false });
	}

	bindEvents() {
		// Закрытие по клику на backdrop
		if (this.options.closeOnBackdrop) {
			this.backdrop.addEventListener('click', () => this.close());
		}

		// Закрытие по кнопке
		if (this.closeBtn) {
			this.closeBtn.addEventListener('click', () => this.close());
		}

		// Закрытие по Escape
		if (this.options.closeOnEscape) {
			this.handleEscape = (e) => {
				if (e.key === 'Escape' && this.state.isOpen) {
					this.close();
				}
			};
			document.addEventListener('keydown', this.handleEscape);
		}
	}

	open(content = '') {
		this.setState({ isOpen: true });

		// Устанавливаем контент
		if (typeof content === 'string') {
			this.body.innerHTML = content;
		} else if (content instanceof HTMLElement) {
			this.body.innerHTML = '';
			this.body.appendChild(content);
		}

		// Показываем модальное окно
		this.element.classList.add('modal--open');
		document.body.classList.add('modal-open');

		// Фокус на модальном окне для accessibility
		this.container.focus();

		// Эмитируем событие
		this.element.dispatchEvent(new CustomEvent('modal:open'));
	}

	close() {
		this.setState({ isOpen: false });

		// Скрываем модальное окно
		this.element.classList.remove('modal--open');
		document.body.classList.remove('modal-open');

		// Эмитируем событие
		this.element.dispatchEvent(new CustomEvent('modal:close'));
	}

	setTitle(title) {
		this.title.textContent = title;
	}

	setFooter(content) {
		if (typeof content === 'string') {
			this.footer.innerHTML = content;
		} else if (content instanceof HTMLElement) {
			this.footer.innerHTML = '';
			this.footer.appendChild(content);
		}
	}

	destroy() {
		// Удаляем глобальные слушатели
		if (this.handleEscape) {
			document.removeEventListener('keydown', this.handleEscape);
		}

		super.destroy();
	}
}
```

### 5.2 Template Literals для рендеринга

**Система рендеринга с template literals:**

```javascript
// TemplateRenderer.js - система рендеринга
class TemplateRenderer {
	// Рендеринг шаблона с данными
	static render(template, data = {}) {
		return template.replace(/\{\{(.+?)\}\}/g, (match, key) => {
			const value = this.getNestedValue(data, key.trim());
			return value !== undefined ? value : match;
		});
	}

	// Получение вложенных значений (например: user.profile.name)
	static getNestedValue(obj, path) {
		return path.split('.').reduce((current, key) => {
			return current && current[key] !== undefined ? current[key] : undefined;
		}, obj);
	}

	// Рендеринг списков
	static renderList(template, items, itemTemplate) {
		if (!Array.isArray(items)) return '';

		const renderedItems = items
			.map((item) => this.render(itemTemplate, item))
			.join('');

		return this.render(template, { items: renderedItems });
	}

	// Условный рендеринг
	static renderIf(condition, template, data = {}) {
		return condition ? this.render(template, data) : '';
	}
}

// Шаблоны компонентов
const templates = {
	// Шаблон карточки товара
	productCard: `
		<article class="card card--product" data-id="{{id}}">
			<div class="card__image-wrapper">
				<img class="card__image" src="{{image}}" alt="{{title}}" loading="lazy">
				{{#if badge}}
					<span class="card__badge">{{badge}}</span>
				{{/if}}
			</div>
			<div class="card__content">
				<h3 class="card__title">{{title}}</h3>
				<p class="card__description">{{description}}</p>
				<div class="card__price">
					<span class="card__price-current">{{price}}</span>
					{{#if oldPrice}}
						<span class="card__price-old">{{oldPrice}}</span>
					{{/if}}
				</div>
				<div class="card__actions">
					<button class="button button--primary" data-action="buy">
						Купить
					</button>
					<button class="button button--outline" data-action="favorite">
						♥
					</button>
				</div>
			</div>
		</article>
	`,

	// Шаблон списка товаров
	productList: `
		<div class="products-grid">
			{{items}}
		</div>
	`,

	// Шаблон кнопки
	button: `
		<button class="button {{#if variant}}button--{{variant}}{{/if}} {{#if size}}button--{{size}}{{/if}}" 
		        {{#if disabled}}disabled{{/if}}
		        {{#if type}}type="{{type}}"{{/if}}>
			{{#if iconLeft}}
				<svg class="button__icon button__icon--left">{{iconLeft}}</svg>
			{{/if}}
			<span class="button__text">{{text}}</span>
			{{#if iconRight}}
				<svg class="button__icon button__icon--right">{{iconRight}}</svg>
			{{/if}}
		</button>
	`,
};

// Расширенный рендерер с поддержкой условий
class AdvancedTemplateRenderer extends TemplateRenderer {
	static render(template, data = {}) {
		// Обрабатываем условные блоки {{#if condition}}...{{/if}}
		let processed = template.replace(
			/\{\{#if\s+(.+?)\}\}([\s\S]*?)\{\{\/if\}\}/g,
			(match, condition, content) => {
				const conditionValue = this.getNestedValue(data, condition.trim());
				return conditionValue ? content : '';
			}
		);

		// Обрабатываем обычные переменные
		return super.render(processed, data);
	}
}
```

**Использование системы рендеринга:**

```javascript
// ProductComponent.js - компонент для отображения товаров
class ProductComponent extends ComponentBase {
	static cssClass = 'product-component';

	get defaultOptions() {
		return {
			template: templates.productCard,
			loadingTemplate: '<div class="loading">Загрузка...</div>',
			errorTemplate: '<div class="error">Ошибка загрузки</div>',
		};
	}

	init() {
		super.init();
		this.setState({ loading: false, error: null, products: [] });
	}

	// Загрузка товаров
	async loadProducts(url) {
		this.setState({ loading: true, error: null });

		try {
			const response = await fetch(url);
			if (!response.ok) throw new Error('Ошибка загрузки');

			const products = await response.json();
			this.setState({ loading: false, products });
		} catch (error) {
			this.setState({ loading: false, error: error.message });
		}
	}

	// Рендеринг компонента
	render() {
		const { loading, error, products } = this.state;

		if (loading) {
			this.element.innerHTML = this.options.loadingTemplate;
			return;
		}

		if (error) {
			this.element.innerHTML = AdvancedTemplateRenderer.render(
				this.options.errorTemplate,
				{ error }
			);
			return;
		}

		// Рендерим список товаров
		const renderedProducts = products
			.map((product) =>
				AdvancedTemplateRenderer.render(this.options.template, product)
			)
			.join('');

		this.element.innerHTML = AdvancedTemplateRenderer.render(
			templates.productList,
			{ items: renderedProducts }
		);

		// Привязываем события после рендеринга
		this.bindProductEvents();
	}

	// События для товаров
	bindProductEvents() {
		this.element.addEventListener('click', (e) => {
			const action = e.target.dataset.action;
			const card = e.target.closest('[data-id]');

			if (!card || !action) return;

			const productId = card.dataset.id;

			switch (action) {
				case 'buy':
					this.handleBuy(productId);
					break;
				case 'favorite':
					this.handleFavorite(productId);
					break;
			}
		});
	}

	handleBuy(productId) {
		// Эмитируем событие покупки
		this.element.dispatchEvent(
			new CustomEvent('product:buy', {
				detail: { productId },
			})
		);
	}

	handleFavorite(productId) {
		// Эмитируем событие добавления в избранное
		this.element.dispatchEvent(
			new CustomEvent('product:favorite', {
				detail: { productId },
			})
		);
	}
}
```

### 5.3 Управление состоянием компонентов

**Простая система управления состоянием:**

```javascript
// StateManager.js - управление глобальным состоянием
class StateManager {
	constructor() {
		this.state = {};
		this.subscribers = {};
	}

	// Получение состояния
	getState(key) {
		return key ? this.state[key] : this.state;
	}

	// Установка состояния
	setState(updates) {
		const prevState = { ...this.state };
		this.state = { ...this.state, ...updates };

		// Уведомляем подписчиков об изменениях
		Object.keys(updates).forEach((key) => {
			if (this.subscribers[key]) {
				this.subscribers[key].forEach((callback) => {
					callback(this.state[key], prevState[key]);
				});
			}
		});
	}

	// Подписка на изменения
	subscribe(key, callback) {
		if (!this.subscribers[key]) {
			this.subscribers[key] = [];
		}
		this.subscribers[key].push(callback);

		// Возвращаем функцию для отписки
		return () => {
			const index = this.subscribers[key].indexOf(callback);
			if (index > -1) {
				this.subscribers[key].splice(index, 1);
			}
		};
	}

	// Асинхронные действия
	async dispatch(action, payload) {
		switch (action) {
			case 'LOAD_PRODUCTS':
				return this.loadProducts(payload);
			case 'ADD_TO_CART':
				return this.addToCart(payload);
			case 'TOGGLE_FAVORITE':
				return this.toggleFavorite(payload);
			default:
				console.warn(`Неизвестное действие: ${action}`);
		}
	}

	async loadProducts(url) {
		this.setState({ productsLoading: true });

		try {
			const response = await fetch(url);
			const products = await response.json();

			this.setState({
				products,
				productsLoading: false,
				productsError: null,
			});
		} catch (error) {
			this.setState({
				productsLoading: false,
				productsError: error.message,
			});
		}
	}

	addToCart(productId) {
		const cart = this.getState('cart') || [];
		const existingItem = cart.find((item) => item.id === productId);

		if (existingItem) {
			existingItem.quantity += 1;
		} else {
			cart.push({ id: productId, quantity: 1 });
		}

		this.setState({ cart: [...cart] });
	}

	toggleFavorite(productId) {
		const favorites = this.getState('favorites') || [];
		const index = favorites.indexOf(productId);

		if (index > -1) {
			favorites.splice(index, 1);
		} else {
			favorites.push(productId);
		}

		this.setState({ favorites: [...favorites] });
	}
}

// Глобальный экземпляр
const appState = new StateManager();
```

**Интеграция компонентов с состоянием:**

```javascript
// CartComponent.js - компонент корзины
class CartComponent extends ComponentBase {
	static cssClass = 'cart-component';

	init() {
		super.init();

		// Подписываемся на изменения корзины
		this.unsubscribeCart = appState.subscribe('cart', () => {
			this.render();
		});

		// Подписываемся на изменения товаров
		this.unsubscribeProducts = appState.subscribe('products', () => {
			this.render();
		});

		this.render();
	}

	render() {
		const cart = appState.getState('cart') || [];
		const products = appState.getState('products') || [];

		// Создаем список товаров в корзине с полной информацией
		const cartItems = cart
			.map((cartItem) => {
				const product = products.find((p) => p.id === cartItem.id);
				return product ? { ...product, quantity: cartItem.quantity } : null;
			})
			.filter(Boolean);

		// Считаем общую сумму
		const total = cartItems.reduce((sum, item) => {
			return sum + parseFloat(item.price.replace(/[^\d]/g, '')) * item.quantity;
		}, 0);

		// Рендерим корзину
		const cartTemplate = `
			<div class="cart">
				<h3 class="cart__title">Корзина ({{itemCount}})</h3>
				<div class="cart__items">
					{{items}}
				</div>
				<div class="cart__total">
					Итого: {{total}} ₽
				</div>
				<button class="button button--primary cart__checkout" 
				        {{#if isEmpty}}disabled{{/if}}>
					Оформить заказ
				</button>
			</div>
		`;

		const cartItemTemplate = `
			<div class="cart__item" data-id="{{id}}">
				<img class="cart__item-image" src="{{image}}" alt="{{title}}">
				<div class="cart__item-info">
					<h4 class="cart__item-title">{{title}}</h4>
					<div class="cart__item-controls">
						<button class="cart__item-decrease" data-action="decrease">-</button>
						<span class="cart__item-quantity">{{quantity}}</span>
						<button class="cart__item-increase" data-action="increase">+</button>
						<button class="cart__item-remove" data-action="remove">×</button>
					</div>
				</div>
				<div class="cart__item-price">{{itemTotal}} ₽</div>
			</div>
		`;

		const renderedItems = cartItems
			.map((item) => {
				const itemTotal =
					parseFloat(item.price.replace(/[^\d]/g, '')) * item.quantity;
				return AdvancedTemplateRenderer.render(cartItemTemplate, {
					...item,
					itemTotal: itemTotal.toLocaleString(),
				});
			})
			.join('');

		this.element.innerHTML = AdvancedTemplateRenderer.render(cartTemplate, {
			itemCount: cartItems.length,
			items: renderedItems,
			total: total.toLocaleString(),
			isEmpty: cartItems.length === 0,
		});
	}

	bindEvents() {
		this.element.addEventListener('click', (e) => {
			const action = e.target.dataset.action;
			const item = e.target.closest('[data-id]');

			if (!item || !action) return;

			const productId = item.dataset.id;

			switch (action) {
				case 'increase':
					appState.addToCart(productId);
					break;
				case 'decrease':
					this.decreaseQuantity(productId);
					break;
				case 'remove':
					this.removeFromCart(productId);
					break;
			}
		});
	}

	decreaseQuantity(productId) {
		const cart = appState.getState('cart') || [];
		const item = cart.find((item) => item.id === productId);

		if (item && item.quantity > 1) {
			item.quantity -= 1;
			appState.setState({ cart: [...cart] });
		} else {
			this.removeFromCart(productId);
		}
	}

	removeFromCart(productId) {
		const cart = appState.getState('cart') || [];
		const updatedCart = cart.filter((item) => item.id !== productId);
		appState.setState({ cart: updatedCart });
	}

	destroy() {
		// Отписываемся от изменений состояния
		if (this.unsubscribeCart) this.unsubscribeCart();
		if (this.unsubscribeProducts) this.unsubscribeProducts();

		super.destroy();
	}
}
```
